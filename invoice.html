Nessun ordine registrato.

`; return; } container.innerHTML = orders.map((o, idx) => `
Fattura n. ${o.number} (${o.id}) €${Number(o.total || 0).toFixed(2)}

Cliente: ${o.name} (${o.email}) - Pagamento: ${o.gateway}

`).join(''); } function viewInvoiceModal(idx) { let orders = []; try { orders = JSON.parse(localStorage.getItem('ls3d_orders')) || []; } catch (e) { orders = []; } const o = orders[idx]; if (!o) return; document.getElementById('inv-modal-number').innerText = `Fattura N. ${o.number}`; document.getElementById('inv-modal-date').innerText = `Data: ${o.date || 'Data odierna'}`; document.getElementById('inv-modal-client').innerText = o.name; document.getElementById('inv-modal-email').innerText = o.email; document.getElementById('inv-modal-address').innerText = o.address || 'Indirizzo non specificato'; document.getElementById('inv-modal-total').innerText = `€${Number(o.total || 0).toFixed(2)}`; const itemsContainer = document.getElementById('inv-modal-items'); itemsContainer.innerHTML = (o.items || []).map(item => `

${item.title}

Colore: ${item.color || 'Standard'}

${item.quantity || item.qty || 1} €${(((Number(item.price) || 0) * (Number(item.quantity || item.qty) || 1))).toFixed(2)} `).join(''); document.getElementById('invoice-modal').classList.remove('hidden'); } function closeInvoiceModal() { document.getElementById('invoice-modal').classList.add('hidden'); } function sendInvoiceEmail(idx) { let orders = []; try { orders = JSON.parse(localStorage.getItem('ls3d_orders')) || []; } catch (e) { orders = []; } const o = orders[idx]; if (!o) return; const orderSummaryText = (o.items || []).map(i => `${i.qty || 1}x ${i.title} (${i.color || 'Standard'}) - €${((i.price || 0) * (i.qty || 1)).toFixed(2)}`).join('\n'); const emailParams = { to_email: o.email, customer_name: o.name, customer_address: o.address, order_id: o.id, order_total: Number(o.total || 0).toFixed(2), order_items: orderSummaryText, payment_gateway: o.gateway }; if (typeof emailjs !== 'undefined') { emailjs.send("YOUR_SERVICE_ID", "YOUR_TEMPLATE_ID", emailParams) .then(function(response) { alert(`Email inviata con successo a ${o.email}!`); }, function(error) { alert("Errore durante l'invio dell'email.
