// payment.js - Gestione PayPal e invio email automatico con BCC a Luigi Schiavello const PAYPAL_EMAIL = "unicobardo@gmail.com"; const ADMIN_EMAIL = "luigi.schiavello@gmail.com"; let selectedGateway = 'paypal'; function selectPaymentType(type) { selectedGateway = type; const isPayPal = type === 'paypal'; document.getElementById('box-opt-paypal').className = isPayPal ? "p-3 bg-blue-950/30 border-2 border-blue-500 rounded-xl cursor-pointer flex items-center justify-between" : "p-3 bg-gray-900 border border-gray-700 rounded-xl cursor-pointer flex items-center justify-between opacity-70"; document.getElementById('box-opt-card').className = !isPayPal ? "p-3 bg-blue-950/30 border-2 border-blue-500 rounded-xl cursor-pointer flex items-center justify-between" : "p-3 bg-gray-900 border border-gray-700 rounded-xl cursor-pointer flex items-center justify-between opacity-70"; document.getElementById('card-inputs-container').classList.toggle('hidden', isPayPal); } function handlePayPalCheckout(orderData) { const totalAmount = orderData.total.toFixed(2); // Salvataggio temporaneo ordine sessionStorage.setItem('pending_order', JSON.stringify(orderData)); // Apertura pagina PayPal ufficiale window.open(`https://www.paypal.com/paypalme/unicobardo/${totalAmount}EUR`, '_blank'); // Mostra la sezione per sbloccare la fattura e inviare l'email automatica showDownloadSection(orderData); } function handleCardCheckout(orderData) { alert("Il pagamento con carta di credito sarà presto attivo. Procedi momentaneamente con PayPal."); } function showDownloadSection(orderData) { document.getElementById('checkout-modal').classList.add('hidden'); // INVIA AUTOMATICAMENTE L'EMAIL SOLO SE L'ORDINE RISULTA PAGATO if (orderData.status === "Pagato") { sendAutomaticPaidEmail(orderData); } const successDiv = document.createElement('div'); successDiv.id = 'success-download-modal'; successDiv.className = 'fixed inset-0 z-50 bg-black/80 backdrop-blur-md flex items-center justify-center p-4'; successDiv.innerHTML = `
✓

Pagamento Confermato!

Il pagamento su PayPal a ${PAYPAL_EMAIL} è stato avviato. Ti è stata inviata una conferma via email (con copia nascosta a Luigi).

`; document.body.appendChild(successDiv); } function sendAutomaticPaidEmail(orderData) { const subject = encodeURIComponent(`Conferma Ordine & Fattura ${orderData.number} - LS3Dmaker`); const itemsText = orderData.items.map(i => `- ${i.title} (Qt: ${i.qty}) - €${(i.price * i.qty).toFixed(2)}`).join('\n'); const body = encodeURIComponent( `Gentile ${orderData.name},\n\n` + `Grazie per il tuo acquisto su LS3Dmaker! Il pagamento tramite PayPal è stato completato con successo.\n\n` + `Dettagli dell'ordine ${orderData.id}:\n` + `${itemsText}\n` + `Spedizione: €${orderData.shipping.toFixed(2)}\n` + `Totale Complessivo: €${orderData.total.toFixed(2)}\n\n` + `Indirizzo di spedizione:\n${orderData.address}\n\n` + `Il tuo ordine è ora in fase di elaborazione/stampa 3D. Ti invieremo il codice di tracking non appena verrà spedito.\n\n` + `Cordiali saluti,\nLS3Dmaker` ); // Usa il parametro &bcc= per mettere il tuo indirizzo in copia nascosta const mailtoUrl = `mailto:${orderData.email}?bcc=${ADMIN_EMAIL}&subject=${subject}&body=${body}`; // Apre silenziosamente il client di posta per l'invio const iframe = document.createElement('iframe'); iframe.style.display = 'none'; iframe.src = mailtoUrl; document.body.appendChild(iframe); setTimeout(() => iframe.remove(), 2000); }
